Алокатор на Zig для Linux

Іноді появляється бажання поглибити свої знання в деякій області, саме це в мене і сталось, а об’ктом дії стала алокація пам’яті та алокатори загалом.

Цей алокатор дуже простий, він має масив розмірів алокацій — кластерів від 1 до 32768 у степенях 2 для об’єднаного списку(Linked List) що вказує на заголовок сторінки. Вміст сторінки включає заголовок та тіло, де заголовок містить посилання на наступну та попередню сторінку та деяку додаткову інформацію, а тіло — блоки по розміру своєї категорії(наприклад сторінка в категорії 32 біт, тому розмір блоку буде 32 біт). Всі алокації що не поміщаються у будь-який кластер виконується напряму, тобто вертається ціла сторінка пам’ті.

Це був не остаточний дизайн програми, до нього була версія що використовувала динамічні розміри блоків у сторінках, де я жонглював вказівниками поки щось не ломалось(нащастя, нічого не ломалось). Я відмовився від нього з однієї простої причини — продуктивність. В мене був цикл в циклі ще з одним циклом, що не давало ефективно алокувати пам’ять. У результаті, нова версія виявилась ефективніше у 3–5 рази за попередню і була набагато безпечніше.

Нажаль, в мене не дійшли руки щоб зробити більше оптимізацій і покращити фрагментацію пам’яті, але на мою думку все що потрібно було зробити — було зроблено.

const std = @import("std");

const MyAllocator = @This();

const BrokenLinkedList = struct {
    pub const Node = struct {
        prev: ?*BrokenLinkedList.Node,
        next: ?*BrokenLinkedList.Node,
    };

    pub const init = BrokenLinkedList{
        .front_node = null,
    };

    front_node: ?*BrokenLinkedList.Node,

    pub fn pushFront(self: *BrokenLinkedList, node: *BrokenLinkedList.Node) void {
        if (self.front_node) |front_node| {
            node.prev = null;
            node.next = front_node;

            front_node.prev = node;

            self.front_node = node;
        } else {
            node.prev = null;
            node.next = null;

            self.front_node = node;
        }
    }

    pub fn popFront(self: *BrokenLinkedList) ?*BrokenLinkedList.Node {
        if (self.front_node) |front_node| {
            if (front_node.next) |next_front_node| {
                next_front_node.prev = null;
            }

            self.front_node = front_node.next;

            front_node.prev = undefined;
            front_node.next = undefined;

            return front_node;
        }

        return null;
    }

    pub fn remove(self: *BrokenLinkedList, node: *BrokenLinkedList.Node) void {
        if (node.prev != null and node.next != null) {
            const prev: *BrokenLinkedList.Node = node.prev.?;
            const next: *BrokenLinkedList.Node = node.next.?;

            prev.next = next;
            next.prev = prev;
        } else if (node.prev) |prev| {
            prev.next = null;

            // self.back_node = prev;
        } else if (node.next) |next| {
            next.prev = null;

            self.front_node = next;
        } else {
            self.front_node = null;
            // self.back_node = null;
        }

        node.prev = undefined;
        node.next = undefined;
    }
};

const Cell = struct {
    list_node: BrokenLinkedList.Node,
};

const Page = struct {
    pub const cells_byte_offset = std.mem.alignForward(usize, @sizeOf(Page), @alignOf(Cell));

    list_node: BrokenLinkedList.Node,

    free_cells_list: BrokenLinkedList,
    full_cells_list: BrokenLinkedList,
};

const SizeClass = struct {
    free_pages_list: BrokenLinkedList = .init,
    full_pages_list: BrokenLinkedList = .init,
};

page_size: usize,

size_classes: [16]SizeClass = .{
    .{}, // 1 << 0
    .{}, // 1 << 1
    .{}, // 1 << 2
    .{}, // 1 << 3
    .{}, // 1 << 4
    .{}, // 1 << 5
    .{}, // 1 << 6
    .{}, // 1 << 7
    .{}, // 1 << 8
    .{}, // 1 << 9
    .{}, // 1 << 10
    .{}, // 1 << 11
    .{}, // 1 << 12
    .{}, // 1 << 13
    .{}, // 1 << 14
    .{}, // 1 << 15
},

pub fn init() MyAllocator {
    const page_size = std.heap.pageSize();

    if (Page.cells_byte_offset >= page_size) {
        unreachable;
    }

    return .{
        .page_size = page_size,
    };
}

pub fn allocator(self: *MyAllocator) std.mem.Allocator {
    return .{
        .ptr = self,
        .vtable = &.{
            .alloc = &alloc,
            .resize = &resize,
            .remap = &remap,
            .free = &free,
        },
    };
}

fn alloc(context: *anyopaque, byte_count: usize, alignment: std.mem.Alignment, return_address: usize) ?[*]u8 {
    _ = return_address;

    const self: *MyAllocator = @ptrCast(@alignCast(context));

    const cell_data_byte_offset = alignment.forward(@sizeOf(Cell));
    const cell_byte_size = std.mem.alignForward(usize, cell_data_byte_offset + byte_count, @alignOf(Cell));

    const page_cells_byte_size = self.page_size - Page.cells_byte_offset;

    const size_class_bytes = std.math.ceilPowerOfTwo(usize, cell_byte_size) catch unreachable;
    const size_class_index = std.math.log2(size_class_bytes);
    const size_class_cells_count = page_cells_byte_size / size_class_bytes;

    if (size_class_cells_count <= 1 or size_class_index >= self.size_classes.len) {
        if (alignment.toByteUnits() > self.page_size) {
            return null;
        }

        const mapping_length = std.mem.alignForward(usize, byte_count, self.page_size);

        const prot: std.posix.PROT = .{ .READ = true, .WRITE = true };
        const flags: std.posix.MAP = .{ .TYPE = .PRIVATE, .ANONYMOUS = true };
        const mapped_memory = std.posix.mmap(null, mapping_length, prot, flags, -1, 0) catch return null;

        return mapped_memory.ptr;
    }

    const size_class = &self.size_classes[size_class_index];

    if (size_class.free_pages_list.front_node == null) {
        const prot: std.posix.PROT = .{ .READ = true, .WRITE = true };
        const flags: std.posix.MAP = .{ .TYPE = .PRIVATE, .ANONYMOUS = true };
        const mapped_memory = std.posix.mmap(null, self.page_size, prot, flags, -1, 0) catch return null;

        const page: *Page = @ptrCast(@alignCast(mapped_memory.ptr));

        page.list_node = undefined;

        page.free_cells_list = .init;
        page.full_cells_list = .init;

        const page_cells_address = @intFromPtr(page) + Page.cells_byte_offset;

        var cell_index: usize = 0;

        while (cell_index < size_class_cells_count) : (cell_index += 1) {
            const cell: *Cell = @ptrFromInt(page_cells_address + cell_index * size_class_bytes);

            cell.list_node = undefined;

            page.free_cells_list.pushFront(&cell.list_node);
        }

        size_class.free_pages_list.pushFront(&page.list_node);
    }

    const free_page_node = size_class.free_pages_list.front_node.?;
    const free_page: *Page = @fieldParentPtr("list_node", free_page_node);

    const free_cell_node = free_page.free_cells_list.popFront().?;

    free_page.full_cells_list.pushFront(free_cell_node);

    if (free_page.free_cells_list.front_node == null) {
        _ = size_class.free_pages_list.popFront();
        size_class.full_pages_list.pushFront(free_page_node);
    }

    const free_cell: *Cell = @fieldParentPtr("list_node", free_cell_node);

    return @ptrFromInt(@intFromPtr(free_cell) + cell_data_byte_offset);
}

fn resize(context: *anyopaque, memory: []u8, alignment: std.mem.Alignment, new_length: usize, return_address: usize) bool {
    _ = return_address;

    const self: *MyAllocator = @ptrCast(@alignCast(context));

    const cell_data_byte_offset = alignment.forward(@sizeOf(Cell));
    const cell_byte_size = std.mem.alignForward(usize, cell_data_byte_offset + memory.len, @alignOf(Cell));

    const page_cells_byte_size = self.page_size - Page.cells_byte_offset;

    const size_class_bytes = std.math.ceilPowerOfTwo(usize, cell_byte_size) catch unreachable;
    const size_class_index = std.math.log2(size_class_bytes);
    const size_class_cells_count = page_cells_byte_size / size_class_bytes;

    if (size_class_cells_count <= 1 or size_class_index >= self.size_classes.len) {
        const mapping_length = std.mem.alignForward(usize, memory.len, self.page_size);
        const new_mapping_length = std.mem.alignForward(usize, new_length, self.page_size);

        return mapping_length == new_mapping_length;
    }

    const new_cell_byte_size = std.mem.alignForward(usize, cell_data_byte_offset + new_length, @alignOf(Cell));
    const new_size_class_bytes = std.math.ceilPowerOfTwo(usize, new_cell_byte_size) catch unreachable;

    return size_class_bytes == new_size_class_bytes;
}

fn remap(context: *anyopaque, memory: []u8, alignment: std.mem.Alignment, new_length: usize, return_address: usize) ?[*]u8 {
    if (resize(context, memory, alignment, new_length, return_address)) {
        return memory.ptr;
    }

    return null;
}

fn free(context: *anyopaque, memory: []u8, alignment: std.mem.Alignment, return_address: usize) void {
    _ = return_address;

    const self: *MyAllocator = @ptrCast(@alignCast(context));

    const cell_data_byte_offset = alignment.forward(@sizeOf(Cell));
    const cell_byte_size = std.mem.alignForward(usize, cell_data_byte_offset + memory.len, @alignOf(Cell));

    const page_cells_byte_size = self.page_size - Page.cells_byte_offset;

    const size_class_bytes = std.math.ceilPowerOfTwo(usize, cell_byte_size) catch unreachable;
    const size_class_index = std.math.log2(size_class_bytes);
    const size_class_cells_count = page_cells_byte_size / size_class_bytes;

    if (size_class_cells_count <= 1 or size_class_index >= self.size_classes.len) {
        const mapping_length = std.mem.alignForward(usize, memory.len, self.page_size);

        const mapped_memory: []align(std.heap.page_size_min) u8 = @alignCast(memory.ptr[0..mapping_length]);

        std.posix.munmap(mapped_memory);

        return;
    }

    const size_class = &self.size_classes[size_class_index];

    const page_address = std.mem.alignBackward(usize, @intFromPtr(memory.ptr), self.page_size);
    const page: *Page = @ptrFromInt(page_address);
    const cell: *Cell = @ptrFromInt(@intFromPtr(memory.ptr) - cell_data_byte_offset);

    if (page.free_cells_list.front_node == null) {
        size_class.full_pages_list.remove(&page.list_node);
        size_class.free_pages_list.pushFront(&page.list_node);
    }

    page.full_cells_list.remove(&cell.list_node);
    page.free_cells_list.pushFront(&cell.list_node);

    if (page.full_cells_list.front_node == null) {
        size_class.free_pages_list.remove(&page.list_node);

        std.posix.munmap(@as([*]align(std.heap.page_size_min) u8, @ptrFromInt(page_address))[0..self.page_size]);
    }
}

pub const Type = f64;

pub const min_coefficient = 3.0 / 4.0;
pub const max_coefficient = 3.0 / 2.0;

pub const loves_coefficient = 0.88;
pub const likes_coefficient = 0.94;
pub const neutral_coefficient = 1.0;
pub const dislikes_coefficient = 1.06;
pub const hates_coefficient = 1.12;

pub const loney_princess_neighbor_count = 1;
pub const loney_princess_coefficient = max_coefficient;

pub const solitude_bonus_coefficient = 0.95;
pub const solitude_bonus_neighbor_npc_count = 2;

pub const crowded_penalty_coefficient = 1.05;
pub const crowded_penalty_neighbor_npc_count = 4;

pub const lovestruck_potion_coefficient = 0.9;

pub fn normalize(coefficient: Type) Type {
    return @max(min_coefficient, @min(coefficient, max_coefficient));
}

pub fn princessCoefficient(neighbor_npc_count: usize) Type {
    if (neighbor_npc_count <= loney_princess_neighbor_count) {
        return loney_princess_coefficient;
    } else {
        return neutral_coefficient;
    }
}

pub fn townCoefficient(neighbor_npc_count: usize) Type {
    if (neighbor_npc_count <= solitude_bonus_neighbor_npc_count) {
        return solitude_bonus_coefficient;
    } else if (neighbor_npc_count >= crowded_penalty_neighbor_npc_count) {
        const multiplication_number = (neighbor_npc_count - crowded_penalty_neighbor_npc_count) + 1;

        var coefficient: Type = neutral_coefficient;

        for (0..multiplication_number) |_| {
            coefficient *= crowded_penalty_coefficient;
        }

        return coefficient;
    } else {
        return neutral_coefficient;
    }
}

pub fn biomeCoefficient(npc_id: npc.Id, biome_id: biome.Id) Type {
    const location_preference = Preference.getLocationPreference(npc_id);

    if (location_preference.loves == biome_id) {
        return loves_coefficient;
    } else if (location_preference.likes == biome_id) {
        return likes_coefficient;
    } else if (location_preference.dislikes == biome_id) {
        return dislikes_coefficient;
    } else if (location_preference.hates == biome_id) {
        return hates_coefficient;
    } else {
        return neutral_coefficient;
    }
}

pub fn neighborsCoefficient(npc_id: npc.Id, neighbor_npc_bits: npc.bits.Type) Type {
    var coefficient: Type = neutral_coefficient;

    const neighbors_preference = Preference.getNeighborsPreference(npc_id);

    const loved_npc_bits = neighbors_preference.loves & neighbor_npc_bits;
    const loved_npc_count = @popCount(loved_npc_bits);

    for (0..loved_npc_count) |_| {
        coefficient *= loves_coefficient;
    }

    const liked_npc_bits = neighbors_preference.likes & neighbor_npc_bits;
    const liked_npc_count = @popCount(liked_npc_bits);

    for (0..liked_npc_count) |_| {
        coefficient *= likes_coefficient;
    }

    const disliked_npc_bits = neighbors_preference.dislikes & neighbor_npc_bits;
    const disliked_npc_count = @popCount(disliked_npc_bits);

    for (0..disliked_npc_count) |_| {
        coefficient *= dislikes_coefficient;
    }

    const hated_npc_bits = neighbors_preference.hates & neighbor_npc_bits;
    const hated_npc_count = @popCount(hated_npc_bits);

    for (0..hated_npc_count) |_| {
        coefficient *= hates_coefficient;
    }

    return coefficient;
}

pub fn lovestruckPotionCoefficient(is_lovestruck_potion: bool) Type {
    if (is_lovestruck_potion) {
        return lovestruck_potion_coefficient;
    } else {

    }

typedef uint8_t u8;
typedef uint64_t u64;

typedef u8 status_t;

static u64 power(u64 number, u64 exponent) {
    if (exponent == 0) {
        return 1;
    }

    u64 result = 1;

    for (u64 index = 0; index < exponent; index += 1) {
        result = result * number;
    }

    return result;
}

static status_t binary_digit_to_number(char digit, u64 *value) {
    if (digit == '0' || digit == '1') {
        *value = digit - '0';

        return SUCCESS;;
    }

    return ERROR_INVALID_VALUE;
}

static status_t decimal_digit_to_number(char digit, u64 *value) {
    if (digit >= '0' && digit <= '9') {
        *value = digit - '0';

        return SUCCESS;
    }

    return ERROR_INVALID_VALUE;
}

static status_t hex_digit_to_number(char digit, u64 *value) {
    if (digit >= '0' && digit <= '9') {
        *value = digit - '0';

        return SUCCESS;
    }

    if (digit >= 'a' && digit <= 'f') {
        *value = digit - 'a' + 10;

        return SUCCESS;
    }

    if (digit >= 'A' && digit <= 'F') {
        *value = digit - 'A' + 10;

        return SUCCESS;
    }

    return ERROR_INVALID_VALUE;
}

static status_t digit_to_number(base_t base, char digit, u64 *value) {
    switch (base) {
        case BASE_BINARY:
            return binary_digit_to_number(digit, value);
        case BASE_DECIMAL:
            return decimal_digit_to_number(digit, value);
        case BASE_HEXADECIMAL:
            return hex_digit_to_number(digit, value);
        default:
            return ERROR_INVALID_VALUE;
    }
}

static status_t parse_number(base_t base, char *string, size_t length, u64 *value) {
    u64 result = 0;

    for (size_t index = 0; index < length; index += 1) {
        char digit = string[index];
        u64 number;

        status_t status = digit_to_number(base, digit, &number);

        if (status != SUCCESS) {
            return status;
        }

        result += number * power(base, length - index - 1);
    }

    *value = result;

    return SUCCESS;
}

int main(void) {
    char buffer[BUFFER_COUNT];

    if (scanf("%s", buffer) < 0) {
        fprintf(stderr, "error: Failed to read from stdin\n");
        return EXIT_FAILURE;
    }

    u64 value = 0;
    size_t buffer_length = strlen(buffer);

    status_t status = parse_number(BASE_BINARY, buffer, buffer_length, &value);

    switch (status) {
        case ERROR_INVALID_VALUE:
            fprintf(stderr, "error: Invalid value\n");
            return EXIT_FAILURE;
        default:
            break;
    }

    printf("value: %"PRIu64"\n", value);

    return EXIT_SUCCESS;
}

static char const upper_letters_start = 'A';

static char const lower_letters_number = lower_letters_end - lower_letters_start;

static char const letters_start = upper_letters_start;
static char const letters_end = lower_letters_end;
static char const letters_number = letters_end - letters_start;

static char const special_symbols_start = upper_letters_end;
static char const special_symbols_end = lower_letters_start;
static char const special_symbols_number = special_symbols_end - special_symbols_start;

static char const normalized_letters_number = letters_number - special_symbols_number;

short smod(short const divisor, short const divident) {
    short mod = divisor % divident;

    if (mod < 0) {
        return mod + divident;
    } else {
        return mod;
    }
}

char *encrypt(char const *const plaintext, size_t const plaintext_length, char letter_shift_width) {
    char *const cyphertext = malloc(plaintext_length + 1);

    if (cyphertext == NULL) {
        return NULL;
    }

    for (size_t symbol_index = 0; symbol_index < plaintext_length; symbol_index += 1) {
        char const symbol = plaintext[symbol_index];

        char const is_lower_case_letter = symbol >= lower_letters_start && symbol < lower_letters_end;
        char const is_upper_case_letter = symbol >= upper_letters_start && symbol < upper_letters_end;

        if (!is_lower_case_letter && !is_upper_case_letter) {
            cyphertext[symbol_index] = symbol;

            continue;
        }

        char shifted_letter = symbol;

        if (shifted_letter >= special_symbols_start) {
            shifted_letter = shifted_letter - special_symbols_number;
        }

        shifted_letter -= letters_start;

        shifted_letter = (char)smod((short)shifted_letter + letter_shift_width, normalized_letters_number);

        shifted_letter += letters_start;

        if (shifted_letter >= special_symbols_start) {
            shifted_letter += special_symbols_number;
        }

        cyphertext[symbol_index] = shifted_letter;
    }

    cyphertext[plaintext_length] = '\0';

    return cyphertext;
}
